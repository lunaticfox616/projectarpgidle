 function clampAim(source,aim){
  if(!validCell(source)||!aim||!Number.isFinite(aim.gx)||!Number.isFinite(aim.gy))return null;
  if(inCastRange(source,aim))return cell(aim);
  // Bound extreme input before squaring, then search the at-most72 legal cells.

  for(const c of castCells(source)){const distance=(c.gx-ax)**2+(c.gy-ay)**2,sourceDistance=(c.gx-source.gx)**2+(c.gy-source.gy)**2;if(distance<bestDistance||distance===bestDistance&&sourceDistance<bestSourceDistance){best=c;bestDistance=distance;bestSourceDistance=sourceDistance;}}
  return best;
